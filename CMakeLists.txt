cmake_minimum_required(VERSION 3.16)
project(irp_emulator VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EMULATOR_SANITIZE "Enable address and undefined behaviour checks" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(EMULATOR_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(emulator_core src/core/model.cpp src/core/scheduler.cpp src/core/memory.cpp)
target_include_directories(emulator_core PUBLIC include)
add_executable(emulator src/host/main.cpp src/host/io.cpp src/host/export.cpp)
target_link_libraries(emulator PRIVATE emulator_core)
enable_testing()
add_executable(core_tests tests/core_tests.cpp tests/allocation_guard.cpp)
target_link_libraries(core_tests PRIVATE emulator_core)
add_test(NAME core COMMAND core_tests)
